<script>
(function() {
  let current = 0, correct = 0;
  const opdrachtEl = document.getElementById('z-opdracht');
  const gebiedEl = document.getElementById('z-gebied');
  const feedbackEl = document.getElementById('z-feedback');
  const scoreEl = document.getElementById('z-score');
  const vraagnrEl = document.getElementById('z-vraagnr');
  const klaarEl = document.getElementById('z-klaar');

  function genSom() {
    const mode = Math.random() > 0.5 ? 'delers' : 'babylon';

    if (mode === 'delers') {
      const delers = [2, 3, 4, 5, 6, 10, 12, 15, 20, 30];
      const deler = delers[Math.floor(Math.random() * delers.length)];
      const antwoord = 60 / deler;
      const verkeerd = [antwoord + 5, antwoord - 5, antwoord * 2].filter(x => x > 0);
      const opties = [antwoord, ...verkeerd].sort(() => Math.random() - 0.5);

      opdrachtEl.innerHTML = 'Hoeveel is <strong style="color: #dc2626;">60 ÷ ' + deler + '</strong>?';
      gebiedEl.innerHTML = '';
      opties.forEach((opt) => {
        const btn = document.createElement('button');
        btn.textContent = opt;
        btn.style.cssText = 'padding: 10px 20px; border-radius: 8px; border: 2px solid #e2e8f0; background: white; cursor: pointer; font-size: 1rem; margin: 4px;';
        btn.onmouseenter = () => { if (!btn.disabled) btn.style.borderColor = '#dc2626'; };
        btn.onmouseleave = () => { if (!btn.disabled) btn.style.borderColor = '#e2e8f0'; };
        btn.onclick = () => {
          Array.from(gebiedEl.children).forEach(b => { b.disabled = true; b.style.opacity = '0.5'; });
          btn.style.opacity = '1';
          const isGoed = opt === antwoord;
          if (isGoed) {
            btn.style.borderColor = '#22c55e'; btn.style.background = '#dcfce7';
            correct++;
            feedbackEl.innerHTML = '✅ Goed! 60 ÷ ' + deler + ' = <strong>' + antwoord + '</strong>.';
            feedbackEl.style.cssText = 'margin-top: 12px; padding: 10px; border-radius: 8px; display: block; background: #dcfce7; color: #166534;';
          } else {
            btn.style.borderColor = '#ef4444'; btn.style.background = '#fee2e2';
            feedbackEl.innerHTML = '❌ Niet goed. 60 ÷ ' + deler + ' = <strong>' + antwoord + '</strong>.';
            feedbackEl.style.cssText = 'margin-top: 12px; padding: 10px; border-radius: 8px; display: block; background: #fee2e2; color: #991b1b;';
          }
          scoreEl.textContent = correct;
          current++;
          vraagnrEl.textContent = current + 1;
          nextStep();
        };
        gebiedEl.appendChild(btn);
      });
    } else {
      const getal = Math.floor(Math.random() * 50 + 10);
      const tientallen = Math.floor(getal / 10);
      const eenheden = getal % 10;
      let babylonVis = '';
      for (let i = 0; i < tientallen; i++) babylonVis += '&lt;';
      for (let i = 0; i < eenheden; i++) babylonVis += 'Y';
      const verkeerd = [getal + 5, getal - 5, getal + 10].filter(x => x > 0);
      const opties = [getal, ...verkeerd].sort(() => Math.random() - 0.5);

      opdrachtEl.innerHTML = 'Wat is dit Babylonische getal?<br><strong style="font-size: 1.5rem; color: #dc2626; font-family: monospace;">' + babylonVis + '</strong><br><span style="font-size: 0.8rem; color: #64748b;">&lt; = 10, Y = 1</span>';
      gebiedEl.innerHTML = '';
      opties.forEach((opt) => {
        const btn = document.createElement('button');
        btn.textContent = opt;
        btn.style.cssText = 'padding: 10px 20px; border-radius: 8px; border: 2px solid #e2e8f0; background: white; cursor: pointer; font-size: 1rem; margin: 4px;';
        btn.onmouseenter = () => { if (!btn.disabled) btn.style.borderColor = '#dc2626'; };
        btn.onmouseleave = () => { if (!btn.disabled) btn.style.borderColor = '#e2e8f0'; };
        btn.onclick = () => {
          Array.from(gebiedEl.children).forEach(b => { b.disabled = true; b.style.opacity = '0.5'; });
          btn.style.opacity = '1';
          const isGoed = opt === getal;
          if (isGoed) {
            btn.style.borderColor = '#22c55e'; btn.style.background = '#dcfce7';
            correct++;
            feedbackEl.innerHTML = '✅ Goed! ' + babylonVis + ' = <strong>' + getal + '</strong>.';
            feedbackEl.style.cssText = 'margin-top: 12px; padding: 10px; border-radius: 8px; display: block; background: #dcfce7; color: #166534;';
          } else {
            btn.style.borderColor = '#ef4444'; btn.style.background = '#fee2e2';
            feedbackEl.innerHTML = '❌ Niet goed. ' + babylonVis + ' = <strong>' + getal + '</strong>.';
            feedbackEl.style.cssText = 'margin-top: 12px; padding: 10px; border-radius: 8px; display: block; background: #fee2e2; color: #991b1b;';
          }
          scoreEl.textContent = correct;
          current++;
          vraagnrEl.textContent = current + 1;
          nextStep();
        };
        gebiedEl.appendChild(btn);
      });
    }
    feedbackEl.style.display = 'none';
  }

  function nextStep() {
    if (current >= 10) {
      gebiedEl.innerHTML = '';
      opdrachtEl.innerHTML = '';
      feedbackEl.style.display = 'none';
      klaarEl.style.display = 'block';
      const toetsSection = document.querySelector('section');
      const toetsLock = document.getElementById('toets-lock');
      if (toetsSection) toetsSection.style.display = '';
      if (toetsLock) toetsLock.style.display = 'none';
    } else {
      const nextBtn = document.createElement('button');
      nextBtn.textContent = 'Volgende →';
      nextBtn.style.cssText = 'margin-top: 8px; padding: 6px 16px; background: #dc2626; color: white; border: none; border-radius: 6px; cursor: pointer;';
      nextBtn.onclick = genSom;
      feedbackEl.appendChild(document.createElement('br'));
      feedbackEl.appendChild(nextBtn);
    }
  }

  genSom();
})();
</script>
